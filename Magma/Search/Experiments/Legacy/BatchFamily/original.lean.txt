/-
  Batch certificate for an entire family of three-round impossible
  XOR differentials of Magma.

  Place after the already proved BatchSearch definitions, or import
  Magma.BatchSearch (and its dependencies) in the refactored project.

  No changes to the existing verifiers. No new axioms/sorry/native_decide.
  All Boolean statements remain proof obligations: #eval true alone
  does not constitute a closed Lean proof.
-/

namespace Magma
namespace GProperties
namespace CarryCertificates
namespace BatchFamily

open Magma.GProperties.MultiRoundDDT
open Magma.GProperties.CarryCertificates.BatchSearch

/-! ### 1. One checked base, multiple checked grouped witnesses -/

/-- Check grouped wiring and emptiness independently for EVERY target
    in a list. This does not repeat the shared carry-base verification. -/
def verifyBatchGroupedOnly
    (p : SearchBaseProposal) (targets : List (V 32)) : Bool :=
  targets.all fun aL =>
    verifySearchWiring aL p (proposeWiring aL p)

/-- Self-contained batch certificate. The same shared proposal is reused
    for each requested `aL`; each proposed grouped result is rechecked. -/
structure BatchCertificate where
  base : SearchBase
  proposal : SearchBaseProposal
  targets : List (V 32)

/-- Independent checker for an entire batch: one initial/carry check,
    followed by individual grouped/empty-domain checks. -/
def verifyBatchCertificate (cert : BatchCertificate) : Bool :=
  verifySearchBase cert.base cert.proposal &&
    verifyBatchGroupedOnly cert.proposal cert.targets

/-- Every member of a checked batch has zero three-round XOR-DDT,
    independently of the third round key and right output difference. -/
theorem xorDDT_rounds_three_zero_of_verifiedBatchCertificate
    (cert : BatchCertificate) (aL k₂ bR : V 32)
    (hverified : verifyBatchCertificate cert = true)
    (hmember : aL ∈ cert.targets) :
    xorDDT_rounds 3
      (threeRoundKeys cert.base.k₀ cert.base.k₁ k₂)
      (aL, cert.base.aR) (cert.base.bL, bR) = 0 := by
  unfold verifyBatchCertificate at hverified
  obtain ⟨hbase, hgroup⟩ := Bool.and_eq_true_iff.mp hverified
  unfold verifyBatchGroupedOnly at hgroup
  have hwiring :
      verifySearchWiring aL cert.proposal
        (proposeWiring aL cert.proposal) = true :=
    (List.all_eq_true.mp hgroup) aL hmember
  exact xorDDT_rounds_three_zero_of_verifiedBatchHit
    cert.base cert.proposal aL k₂ bR
    (proposeWiring aL cert.proposal) hbase hwiring

/-! ### 2. A family theorem for all 32 one-hot differences -/

/-- Concrete proposal for the 32 one-bit input-left differences. -/
def exampleOneBitBatch : BatchCertificate :=
  { base := exampleBase
    proposal := proposeSearchBase exampleBase
    targets := oneBitDifferences32 }

private theorem oneBitDifferences32_mem (i : Fin 32) :
    BitVec.ofNat 32 (2 ^ i.val) ∈ oneBitDifferences32 := by
  unfold oneBitDifferences32
  apply List.mem_map.mpr
  exact ⟨i.val, List.mem_range.mpr i.isLt, rfl⟩

/-- One Boolean certificate establishes ALL 32 impossible-differential
    claims, not merely the first `aL = 1` candidate. This theorem is
    conditional until the Boolean check itself is proved equal to true. -/
theorem oneBitFamily_impossible_of_checked
    (hchecked : verifyBatchCertificate exampleOneBitBatch = true)
    (i : Fin 32) (k₂ bR : V 32) :
    xorDDT_rounds 3
      (threeRoundKeys (0#32) (0#32) k₂)
      (BitVec.ofNat 32 (2 ^ i.val), (1#32))
      ((0#32), bR) = 0 := by
  have hmember :
      BitVec.ofNat 32 (2 ^ i.val) ∈ exampleOneBitBatch.targets := by
    change BitVec.ofNat 32 (2 ^ i.val) ∈ oneBitDifferences32
    exact oneBitDifferences32_mem i
  exact xorDDT_rounds_three_zero_of_verifiedBatchCertificate
    exampleOneBitBatch (BitVec.ofNat 32 (2 ^ i.val))
    k₂ bR hchecked hmember

/-! ### 3. Larger exploratory search: the 496 two-bit differences -/

/-- All distinct two-bit masks, indexed by `0 ≤ i < j < 32`.
    This is an UNTRUSTED candidate list, not a cryptographic assertion. -/
def twoBitDifferences32 : List (V 32) :=
  (List.range 32).flatMap fun i =>
    ((List.range 32).filter (fun j => decide (i < j))).map fun j =>
      BitVec.ofNat 32 (2 ^ i + 2 ^ j)

/-- 32 one-bit candidates plus 496 two-bit candidates. -/
def atMostTwoBitDifferences32 : List (V 32) :=
  oneBitDifferences32 ++ twoBitDifferences32

/-- Keep only hits that satisfy BOTH the fast prefilter and the
    independent grouped verifier. The shared base is checked separately. -/
def independentlyVerifiedHits
    (p : SearchBaseProposal) (candidates : List (V 32)) :
    List (V 32) :=
  (scanAL p candidates).filter fun aL =>
    verifySearchWiring aL p (proposeWiring aL p)

private theorem independentlyVerifiedHits_sound
    (p : SearchBaseProposal) (candidates : List (V 32))
    (aL : V 32)
    (h : aL ∈ independentlyVerifiedHits p candidates) :
    verifySearchWiring aL p (proposeWiring aL p) = true := by
  unfold independentlyVerifiedHits at h
  exact (List.mem_filter.mp h).2

/-- A verified hit from a large scan is a genuine three-round impossible
    differential, provided the SHARED carry base passed its verifier. -/
theorem xorDDT_rounds_three_zero_of_independentlyVerifiedHit
    (b : SearchBase) (p : SearchBaseProposal)
    (candidates : List (V 32))
    (aL k₂ bR : V 32)
    (hbase : verifySearchBase b p = true)
    (hmem : aL ∈ independentlyVerifiedHits p candidates) :
    xorDDT_rounds 3 (threeRoundKeys b.k₀ b.k₁ k₂)
      (aL, b.aR) (b.bL, bR) = 0 := by
  exact xorDDT_rounds_three_zero_of_verifiedBatchHit
    b p aL k₂ bR (proposeWiring aL p) hbase
    (independentlyVerifiedHits_sound p candidates aL hmem)

/-! ### 4. Diagnostics: group costs versus base costs -/

/-- Verify ALL 32 grouped candidates in one pass; does NOT rerun the base. -/
def profileAllOneBitGrouped : IO Unit := do
  let p := proposeSearchBase exampleBase
  let t₀ ← IO.monoMsNow
  let ok := verifyBatchGroupedOnly p oneBitDifferences32
  let t₁ ← IO.monoMsNow
  IO.println s!"[family] grouped-only all 32 verified={ok}; time={t₁-t₀}ms"

/-- One expensive shared check and all 32 cheap grouped checks. -/
def profileFullOneBitBatch : IO Unit := do
  let t₀ ← IO.monoMsNow
  let cert := exampleOneBitBatch
  let baseOK := verifySearchBase cert.base cert.proposal
  let t₁ ← IO.monoMsNow
  let groupedOK := verifyBatchGroupedOnly cert.proposal cert.targets
  let t₂ ← IO.monoMsNow
  IO.println s!"[family] base={baseOK}, all-grouped={groupedOK}"
  IO.println s!"[family] base={t₁-t₀}ms, grouped={t₂-t₁}ms"

/-- Explore all two-bit differences with the fast prefilter, then
    verify only prefilter hits (the shared base is NOT rechecked). -/
def profileTwoBitFamily : IO Unit := do
  let p := proposeSearchBase exampleBase
  let candidates := twoBitDifferences32
  let t₀ ← IO.monoMsNow
  let hits := scanAL p candidates
  let t₁ ← IO.monoMsNow
  let checked := independentlyVerifiedHits p candidates
  let t₂ ← IO.monoMsNow
  IO.println s!"[two-bit] candidates={candidates.length}, fast hits={hits.length}"
  IO.println s!"[two-bit] independently verified={checked.length}"
  IO.println s!"[two-bit] fast scan={t₁-t₀}ms; grouped verifier={t₂-t₁}ms"
  IO.println s!"[two-bit] first verified={checked.head?}"

  #eval Magma.GProperties.CarryCertificates.BatchFamily.profileFullOneBitBatch

  For a closed family theorem, it is still necessary to prove:

        (BitVec.ofNat 32 (2 ^ i.val), (1#32)) ((0#32), bR) = 0 := by

    -- is required. `#eval true` is not such a proof; no native_decide.

  `decide` may be costly when certifying a closed example. Keep the

-/
